import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.Scanner;

class Product {

    private int id;
    private String name;
    private double price;

    public Product(int id, String name, double price) {
        this.id = id;
        this.name = name;
        this.price = price;
    }

    public int getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public double getPrice() {
        return price;
    }

    public void display() {
        System.out.println(id + " - " + name + " - ₹" + price);
    }
}

class Cart {

    private ArrayList<Product> products = new ArrayList<>();

    public void addProduct(Product p) {
        products.add(p);
    }

    public double getTotal() {

        double total = 0;

        for (Product p : products) {
            total += p.getPrice();
        }

        return total;
    }

    public void showCart() {

        for (Product p : products) {
            p.display();
        }

        System.out.println("Total: ₹" + getTotal());
    }

    public ArrayList<Product> getProducts() {
        return products;
    }
}

abstract class Payment {

    protected double amount;

    public Payment(double amount) {
        this.amount = amount;
    }

    public abstract void pay();
}

class CreditCardPayment extends Payment {

    private String cardNumber;

    public CreditCardPayment(double amount, String cardNumber) {
        super(amount);
        this.cardNumber = cardNumber;
    }

    @Override
    public void pay() {
        System.out.println("Paid ₹" + amount + " via Credit Card: " + cardNumber);
    }
}

class UPIPayment extends Payment {

    private String upiId;

    public UPIPayment(double amount, String upiId) {
        super(amount);
        this.upiId = upiId;
    }

    @Override
    public void pay() {
        System.out.println("Paid ₹" + amount + " via UPI: " + upiId);
    }
}

class CashPayment extends Payment {

    public CashPayment(double amount) {
        super(amount);
    }

    @Override
    public void pay() {
        System.out.println("Paid ₹" + amount + " in Cash.");
    }
}

class Order {

    private Cart cart;
    private Payment payment;

    public Order(Cart cart, Payment payment) {
        this.cart = cart;
        this.payment = payment;
    }

    public void placeOrder() {

        System.out.println("\n--- Order Summary ---");
        cart.showCart();

        System.out.println("\nProcessing Payment...");
        payment.pay();

        saveOrder();
    }

    private void saveOrder() {

        try {

            FileWriter fw = new FileWriter("orders.txt", true);

            fw.write("Order Total: ₹" + cart.getTotal() + "\n");
            fw.write("----------------------\n");

            fw.close();

            System.out.println("Order saved to file.");

        } catch (IOException e) {

            System.out.println("Error saving order.");
        }
    }
}

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        Cart cart = new Cart();

        Product p1 = new Product(1, "Laptop", 50000);
        Product p2 = new Product(2, "Phone", 20000);
        Product p3 = new Product(3, "Headphones", 2000);

        Product[] products = {p1, p2, p3};

        System.out.println("Available Products:");

        for (Product p : products) {
            p.display();
        }

        while (true) {

            System.out.print("Enter product ID to add (0 to stop): ");
            int id = sc.nextInt();

            if (id == 0) {
                break;
            }

            for (Product p : products) {

                if (p.getId() == id) {
                    cart.addProduct(p);
                }
            }
        }

        cart.showCart();

        System.out.println("\nChoose Payment Method:");
        System.out.println("1. Credit Card");
        System.out.println("2. UPI");
        System.out.println("3. Cash");

        int choice = sc.nextInt();
        sc.nextLine();

        Payment payment;

        if (choice == 1) {

            System.out.print("Enter Card Number: ");
            String card = sc.nextLine();

            payment = new CreditCardPayment(cart.getTotal(), card);

        } else if (choice == 2) {

            System.out.print("Enter UPI ID: ");
            String upi = sc.nextLine();

            payment = new UPIPayment(cart.getTotal(), upi);

        } else {

            payment = new CashPayment(cart.getTotal());
        }

        Order order = new Order(cart, payment);
        order.placeOrder();

        sc.close();
    }
}
